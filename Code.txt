  const schema=getRequestSchema_(sheet),rows=readRequestObjects_(sheet,schema),proposals=rows.map(function(item){

    if(match&&!match.ambiguous){artist=match.artist;title=match.title;}
    return{item:item,artist:artist,title:title,match:match};
  });
  // Reverting one collision can expose another at its original key; settle all keys before writing any names.
  let reverted;
  do{
    reverted=false;const active=new Map();
    proposals.forEach(function(p){if(p.item.status.toLowerCase()==='added')return;const key=baseSongKey_(p.artist,p.title);if(!active.has(key))active.set(key,[]);active.get(key).push(p);});

  }while(reverted);
  let changed=0;
  proposals.forEach(function(p){
    const item=p.item,match=p.match;
    if(match&&match.ambiguous)logCatalogMatch_(ss,'Ambiguous existing request match',item.artist,item.title,match);
    if(p.artist===item.artist&&p.title===item.title)return;
    if(match&&!match.ambiguous&&match.matchType==='close')logCatalogMatch_(ss,'Fuzzy existing request corrected',item.artist,item.title,match);
    if(p.artist!==item.artist){item.artist=p.artist;changed++;}

    writeRequestObject_(sheet,schema,item.row,item);
  });
  if(changed)clearRequestsPublicCache_();return changed;

function isMalformedSongRecord_(artist,title){
  const a=cleanText_(artist),t=cleanText_(title);
  if(!a||!t)return true;
  function hasBrokenCharacter_(text){
    for(let i=0;i<text.length;i++){
      const code=text.charCodeAt(i);
      if(code===0||code===0xFFFD||code===0xFFFC||code===0x25A0||code===0x25A1||code===0x25AF||code===0x25FB||code===0x25FC||(code>=0xE000&&code<=0xF8FF))return true;
    }
    return false;
  }
  if(hasBrokenCharacter_(a)||hasBrokenCharacter_(t))return true;
  if(!/[a-z]/i.test(removeDiacritics_(a)))return true;
  if(/^[><=≤≥]+$/.test(a)||/^[><=≤≥]+$/.test(t))return true;
  
  if(/^g0b\s*[><=≤≥]/i.test(t)||/\bg0b\s*[><=≤≥]/i.test(t))return true;
